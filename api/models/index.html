    class ConstantOccuModel(OccuModel):
    """
    Intercept-only occupancy model with one logit per species.

    Parameters
    ----------
    n_species : int
        Number of species.
    """
    def __init__(self, n_species: int):
        super().__init__(n_species=n_species)

        self.psi_model = ConstantBlock(n_species)
        self.d_model = ConstantBlock(n_species)

    def initialize_rough_estimate(self, data: "OccuData") -> None:
        """
        Initialize intercept logits from rough moment-based estimates.

        Parameters
        ----------
        data : OccuData
            Occupancy dataset used to compute rough estimates.

        Returns
        -------
        None
        """
        y = data.y
        indices_visit_site = data.indices_visit_site # shape: (n_visits,) with values in [0, n_sites-1]

        n_sites = data.n_sites
        n_visits = data.n_visits
        n_species = data.n_species

        n_D = y.sum(dim=0) # shape: (n_species,)
        S = n_sites # total number of sites
        K = n_visits / n_sites # average number of visits per site
        # Number sites with at least one detection for each species

        # Binary indicator of whether each species was detected at each site
        detected = torch.zeros((n_sites, n_species), device=y.device, dtype=y.dtype)
        detected.index_add_(0, indices_visit_site, y)
        detected = detected.bool() # shape: (n_sites, n_species)
        S_D = detected.sum(dim=0) # shape: (n_species,)

        # Solve the equations for the rough estimates of psi and d. 
        # The equations are:
        # hat_d = 1 - (1 - hat_d) ** K * n_D / (S_D * K)
        # hat_psi = S_D / (S * (1 - (1 - hat_d) ** K))

        # If we approximate 1 - (1 - hat_d) ** K by 1
        hat_d = (n_D / (S_D * K)).clamp(min=1e-6, max=1-1e-6)
        hat_psi = (S_D / (S * (1 - (1 - hat_d) ** K))).clamp(min=1e-6, max=1-1e-6)

        with torch.no_grad():
            self.psi_model.logit_species.copy_(torch.logit(hat_psi))
            self.d_model.logit_species.copy_(torch.logit(hat_d))

    class LinearMLPOccuModel(OccuModel):
    """
    Hybrid model combining linear and MLP components.

    Notes
    -----
    For each component (psi and d), the first ``n_cov_*_linear`` columns are
    sent to the linear branch and the next ``n_cov_*_mlp`` columns to the MLP
    branch. Outputs are summed.
    More details in the docstring of ``LinearMLPBlock``.
    """
    def __init__(
            self, 
            n_species: int, 
            n_cov_psi_linear: int = 0,
            n_cov_psi_mlp: int = 0,
            n_cov_d_linear: int = 0,
            n_cov_d_mlp: int = 0,
            hidden_dims_psi: Optional[list[int]] = [], 
            hidden_dims_d: Optional[list[int]] = [],
            activation_fn_psi: Optional[Callable[..., torch.nn.Module]] = nn.ReLU,
            activation_fn_d: Optional[Callable[..., torch.nn.Module]] = nn.ReLU,
            dropout_psi: Optional[float] = 0.0, 
            dropout_d: Optional[float] = 0.0
        ):
        super().__init__(n_species=n_species)

        self.psi_model = LinearMLPBlock(
            n_species=n_species, 
            n_cov_linear=n_cov_psi_linear, 
            n_cov_mlp=n_cov_psi_mlp, 
            hidden_dims_mlp=hidden_dims_psi,
            activation_fn_mlp=activation_fn_psi, 
            dropout_mlp=dropout_psi
        )
        self.d_model = LinearMLPBlock(
            n_species=n_species, 
            n_cov_linear=n_cov_d_linear, 
            n_cov_mlp=n_cov_d_mlp, 
            hidden_dims_mlp=hidden_dims_d,
            activation_fn_mlp=activation_fn_d, 
            dropout_mlp=dropout_d
        )

    class LinearOccuModel(OccuModel):
    """
    Linear occupancy model with species-specific outputs.

    Parameters
    ----------
    n_species : int
        Number of species.
    n_cov_psi : int, default=1
        Number of occupancy covariates.
    n_cov_d : int, default=1
        Number of detection covariates.
    """
    def __init__(self, n_species: int, n_cov_psi: int = 1, n_cov_d: int = 1):
        super().__init__(n_species=n_species)

        # Model occupancy logits
        if n_cov_psi < 0:
            raise ValueError("n_cov_psi must be non-negative.")
        elif n_cov_psi == 0:
            warnings.warn(
                "n_cov_psi=0 results in no covariates for occupancy. Using constant block instead of linear.",
                UserWarning
            )
            self.psi_model = ConstantBlock(n_species)
        else:
            self.psi_model = nn.Linear(n_cov_psi, n_species)

        # Model detection logits
        if n_cov_d < 0:
            raise ValueError("n_cov_d must be non-negative.")
        elif n_cov_d == 0:
            warnings.warn(
                "n_cov_d=0 results in no covariates for detection. Using constant block instead of linear.",
                UserWarning
            )
            self.d_model = ConstantBlock(n_species)
        else:
            self.d_model = nn.Linear(n_cov_d, n_species)

    class MLPOccuModel(OccuModel):
    """
    MLP-based occupancy model for occupancy and detection components.

    Parameters
    ----------
    n_species : int
        Number of species.
    n_cov_psi : int, default=1
        Number of occupancy covariates.
    n_cov_d : int, default=1
        Number of detection covariates.
    hidden_dims_psi : list[int], optional
        Hidden layer sizes for occupancy MLP.
    hidden_dims_d : list[int], optional
        Hidden layer sizes for detection MLP.
    activation_fn_psi : callable, optional
        Activation constructor for occupancy MLP.
    activation_fn_d : callable, optional
        Activation constructor for detection MLP.
    dropout_psi : float, default=0.0
        Dropout probability for occupancy MLP.
    dropout_d : float, default=0.0
        Dropout probability for detection MLP.
    """
    def __init__(
            self, 
            n_species: int, 
            n_cov_psi: int = 1, 
            n_cov_d: int = 1,
            hidden_dims_psi: Optional[list[int]] = [], 
            hidden_dims_d: Optional[list[int]] = [],
            activation_fn_psi: Optional[Callable[..., torch.nn.Module]] = nn.ReLU,
            activation_fn_d: Optional[Callable[..., torch.nn.Module]] = nn.ReLU,
            dropout_psi: Optional[float] = 0.0, 
            dropout_d: Optional[float] = 0.0
        ):
        super().__init__(n_species=n_species)

        # Model occupancy logits
        if n_cov_psi < 0:
            raise ValueError("n_cov_psi must be non-negative.")
        elif n_cov_psi == 0:
            warnings.warn(
                "n_cov_psi=0 results in no covariates for occupancy. Using constant block instead of mlp.",
                UserWarning
            )
            self.psi_model = ConstantBlock(n_species)
        elif hidden_dims_psi == []:
            warnings.warn(
                "hidden_dims_psi is empty, resulting in no hidden layers for occupancy. Using linear layer instead of mlp.",
                UserWarning
            )
            self.psi_model = nn.Linear(n_cov_psi, n_species)
        else:
            self.psi_model = MLPBlock(
                input_dim=n_cov_psi, output_dim=n_species, hidden_dims=hidden_dims_psi, 
                activation_fn=activation_fn_psi, dropout=dropout_psi
            )

        # Model detection logits
        if n_cov_d < 0:
            raise ValueError("n_cov_d must be non-negative.")
        elif n_cov_d == 0:
            warnings.warn(
                "n_cov_d=0 results in no covariates for detection. Using constant block instead of mlp.",
                UserWarning
            )
            self.d_model = ConstantBlock(n_species)
        elif hidden_dims_d == []:
            warnings.warn(
                "hidden_dims_d is empty, resulting in no hidden layers for detection. Using linear layer instead of mlp.",
                UserWarning
            )
            self.d_model = nn.Linear(n_cov_d, n_species)
        else:
            self.d_model = MLPBlock(
                input_dim=n_cov_d, output_dim=n_species, hidden_dims=hidden_dims_d, 
                activation_fn=activation_fn_d, dropout=dropout_d
            )

    class OccuModel(ABC):
    """
    Base occupancy model mapping covariates to occupancy and detection logits.

    Parameters
    ----------
    psi_model : torch.nn.Module or None, default=None
        Module used to predict occupancy logits from site covariates.
    d_model : torch.nn.Module or None, default=None
        Module used to predict detection logits from visit covariates.
    n_species : int, default=1
        Number of species modeled jointly.
    """

    def __init__(
        self,
        psi_model: Optional[nn.Module] = None,
        d_model: Optional[nn.Module] = None,
        n_species: int = 1,
    ):
        super().__init__()

        self.psi_model = psi_model
        self.d_model = d_model
        self.n_species = n_species

    def predict_logit_psi(self, X_psi: Optional[torch.Tensor] = None) -> torch.Tensor:
        """
        Predict occupancy logits.

        Parameters
        ----------
        X_psi : torch.Tensor or None, default=None
            Site covariates with shape ``(n_sites_sub, n_cov_psi)``.
            Can be ``None`` for intercept-only models.

        Returns
        -------
        torch.Tensor
            Occupancy logits with shape ``(n_sites_sub, n_species)`` or
            ``(1, n_species)`` for intercept-only models.

        Raises
        ------
        NotImplementedError
            If ``psi_model`` is not set.
        """
        if self.psi_model is None:
            raise NotImplementedError(
                "Subclasses must implement predict_logit_psi to return occupancy logits."
            )
        return self.psi_model(X_psi)

    def predict_logit_d(self, X_d: Optional[torch.Tensor] = None) -> torch.Tensor:
        """
        Predict detection logits.

        Parameters
        ----------
        X_d : torch.Tensor or None, default=None
            Visit covariates with shape ``(n_visits_sub, n_cov_d)``.
            Can be ``None`` for intercept-only models.

        Returns
        -------
        torch.Tensor
            Detection logits with shape ``(n_visits_sub, n_species)`` or
            ``(1, n_species)`` for intercept-only models.

        Raises
        ------
        NotImplementedError
            If ``d_model`` is not set.
        """
        if self.d_model is None:
            raise NotImplementedError(
                "Subclasses must implement predict_logit_d to return detection logits."
            )
        return self.d_model(X_d)

    def forward_logits(
        self, 
        *, 
        X_psi: Optional[torch.Tensor] = None,
        X_d: Optional[torch.Tensor] = None
    ) -> Tuple[torch.Tensor, torch.Tensor]:
        """
        Compute occupancy and detection logits.

        Parameters
        ----------
        X_psi : torch.Tensor or None, default=None
            Site covariates.
        X_d : torch.Tensor or None, default=None
            Visit covariates.

        Returns
        -------
        tuple[torch.Tensor, torch.Tensor]
            Tuple ``(logit_psi, logit_d)``.
        """
        logit_psi = self.predict_logit_psi(X_psi)
        logit_d = self.predict_logit_d(X_d)
        return logit_psi, logit_d

    def eval(self):
        """
        Set internal modules to evaluation mode.

        Returns
        -------
        None
        """
        if self.psi_model is not None:
            self.psi_model.eval()
        if self.d_model is not None:
            self.d_model.eval()

    def train(self):
        """
        Set internal modules to training mode.

        Returns
        -------
        None
        """
        if self.psi_model is not None:
            self.psi_model.train()
        if self.d_model is not None:
            self.d_model.train()

    def to(self, *args, **kwargs):
        """
        Move internal modules to a target device/dtype.

        Parameters
        ----------
        *args
            Positional arguments forwarded to ``torch.nn.Module.to``.
        **kwargs
            Keyword arguments forwarded to ``torch.nn.Module.to``.

        Returns
        -------
        OccuModel
            The same model instance.
        """
        if self.psi_model is not None:
            self.psi_model.to(*args, **kwargs)
        if self.d_model is not None:
            self.d_model.to(*args, **kwargs)
        return self

    class QualiLinearMLPOccuModel(OccuModel):
    """
    Hybrid model combining qualitative intercept, linear and MLP components.

    Notes
    -----
    For each component (psi and d), the first ``len(n_classes_*_intercept)`` 
    columns are sent to the intercept branch, the next ``n_cov_*_linear`` 
    columns to the linear branch and the next ``n_cov_*_mlp`` columns to the
    MLP branch. Outputs are summed.
    """
    def __init__(
            self, 
            n_species: int, 
            n_classes_psi_intercept: list[int] = [],
            n_cov_psi_linear: int = 0,
            n_cov_psi_mlp: int = 0,
            n_classes_d_intercept: list[int] = [],
            n_cov_d_linear: int = 0,
            n_cov_d_mlp: int = 0,
            hidden_dims_psi: Optional[list[int]] = [], 
            hidden_dims_d: Optional[list[int]] = [],
            activation_fn_psi: Optional[Callable[..., torch.nn.Module]] = nn.ReLU,
            activation_fn_d: Optional[Callable[..., torch.nn.Module]] = nn.ReLU,
            dropout_psi: Optional[float] = 0.0, 
            dropout_d: Optional[float] = 0.0,
            n_fourier_features_psi: Optional[list[int]] = None,
            n_fourier_features_d: Optional[list[int]] = None
        ):
        super().__init__(n_species=n_species)

        self.psi_model = QualiLinearMLPBlock(
            n_species=n_species, 
            n_classes_intercept=n_classes_psi_intercept,
            n_cov_linear=n_cov_psi_linear, 
            n_cov_mlp=n_cov_psi_mlp, 
            hidden_dims_mlp=hidden_dims_psi,
            activation_fn_mlp=activation_fn_psi, 
            dropout_mlp=dropout_psi,
            n_fourier_features=n_fourier_features_psi
        )
        self.d_model = QualiLinearMLPBlock(
            n_species=n_species, 
            n_classes_intercept=n_classes_d_intercept,
            n_cov_linear=n_cov_d_linear, 
            n_cov_mlp=n_cov_d_mlp, 
            hidden_dims_mlp=hidden_dims_d,
            activation_fn_mlp=activation_fn_d, 
            dropout_mlp=dropout_d,
            n_fourier_features=n_fourier_features_d
        )
