    class OccuData():
    """
    Canonical data container for single-season occupancy modeling.

    Parameters
    ----------
    y : torch.Tensor
        Visit-level detections, shape ``(n_visits, n_species)``.
    visit_site_ids : list[int] or list[str]
        Site identifier per visit.
    X_psi : torch.Tensor or None, default=None
        Site covariates, shape ``(n_sites, n_cov_psi)``.
    X_d : torch.Tensor or None, default=None
        Visit covariates, shape ``(n_visits, n_cov_d)``.
    site_ids_in_X_psi : list[int] or list[str] or None, default=None
        Site IDs aligned with rows of ``X_psi``.
    """
    def __init__(
        self,
        *,
        y: torch.Tensor,
        visit_site_ids: list[int] | list[str],
        X_psi: Optional[torch.Tensor] = None,
        X_d: Optional[torch.Tensor] = None,
        site_ids_in_X_psi: Optional[list[int] | list[str]] = None
    ):

        print("Initializing OccuData...")
        self.y = torch.as_tensor(y)
        self.X_psi = torch.as_tensor(X_psi) if X_psi is not None else None
        self.X_d = torch.as_tensor(X_d) if X_d is not None else None

        if not isinstance(visit_site_ids, list):
            raise ValueError(f"visit_site_ids must be a list of site ids (e.g. names), got {type(visit_site_ids)}")

        if not (all([isinstance(e, str) for e in visit_site_ids]) or all([isinstance(e, int) for e in visit_site_ids])):
            raise ValueError("visit_site_ids must be a list of all strings or all integers.")

        if self.y.ndim != 2:
            raise ValueError(f"y must be 2D (n_visits, n_species); got {tuple(self.y.shape)}")
        if len(visit_site_ids) != self.y.shape[0]:
            raise ValueError(
                f"visit_site_ids length {len(visit_site_ids)} must match y n_visits {self.y.shape[0]}"
            )
        if self.X_d is not None:
            if self.X_d.ndim != 2:
                raise ValueError(f"X_d must be 2D (n_visits, n_cov_d); got {tuple(self.X_d.shape)}")
            if self.X_d.shape[0] != self.y.shape[0]:
                raise ValueError(f"X_d first dim ({self.X_d.shape[0]}) must match y n_visits ({self.y.shape[0]}).")
        if self.X_psi is not None:
            if self.X_psi.ndim != 2:
                raise ValueError(f"X_psi must be 2D (n_sites, n_cov_psi); got {tuple(self.X_psi.shape)}")
            if self.X_psi.shape[0] != len(set(visit_site_ids)):
                message = f"X_psi first dim ({self.X_psi.shape[0]}) must match number of unique site ids in visit_site_ids ({len(set(visit_site_ids))})."
                if self.X_psi.shape[0] > len(set(visit_site_ids)):
                    message += "\nIf X_psi has more rows than unique site ids, consider dropping unused rows in X_psi."
                else:
                    message += "\nIf X_psi has fewer rows than unique site ids, consider adding missing rows to X_psi or dropping visits with site ids that don't have corresponding rows in X_psi."
                raise ValueError(message)
        if site_ids_in_X_psi is not None:
            if len(set(site_ids_in_X_psi)) != len(site_ids_in_X_psi):
                raise ValueError("site_ids_in_X_psi must contain unique site ids (no duplicates).")
            if set(site_ids_in_X_psi) != set(visit_site_ids):
                raise ValueError("site_ids_in_X_psi must contain the same unique site ids as visit_site_ids.")
            if self.X_psi is not None:
                if len(site_ids_in_X_psi) != self.X_psi.shape[0]:
                    raise ValueError(f"site_ids_in_X_psi length ({len(site_ids_in_X_psi)}) must match X_psi first dim ({self.X_psi.shape[0]}).")

        # Initialize splitting with all sites in train by default (can be updated later with build_dataloaders)
        self.splitting = {
            "train": list(set(visit_site_ids)),
            "val": [],
            "test": []
        }

        # Usefull dimensions    
        self.n_sites = len(set(visit_site_ids))
        self.n_visits = len(visit_site_ids)
        self.n_species = self.y.shape[1]
        self.n_cov_psi = self.X_psi.shape[1] if self.X_psi is not None else 0
        self.n_cov_d = self.X_d.shape[1] if self.X_d is not None else 0

        # Dictionnary mapping: 
        #  - site ids in visit_site_ids 
        # to 
        # - site indices in X_psi (i.e. 0 to n_sites-1) and the values in indices_visit_site

        # self.mapping_site_idx = {initial_id: idx for idx, initial_id in enumerate(sorted(set(visit_site_ids)))}
        # unique_site_ids = []
        # for site_id in visit_site_ids:
        #     if site_id not in unique_site_ids:
        #         unique_site_ids.append(site_id)
        if site_ids_in_X_psi is not None:
            self.site_ids_in_X_psi = site_ids_in_X_psi
        elif self.X_psi is None:
            self.site_ids_in_X_psi = list(dict.fromkeys(visit_site_ids)) # preserves order of first occurrence
        else: # If site_ids_in_X_psi is None and X_psi is not None
            raise ValueError("If X_psi is provided, site_ids_in_X_psi must also be provided to map site ids to rows in X_psi.")
        self.mapping_site_idx = {initial_id: idx for idx, initial_id in enumerate(self.site_ids_in_X_psi)}

        # Reindex visit_site_ids to site indices in X_psi (i.e. 0 to n_sites-1)
        # indices = [self.mapping_site_idx[site_id] for site_id in visit_site_ids]
        mapping = self.mapping_site_idx.__getitem__
        indices = [mapping(s) for s in visit_site_ids]
        self.indices_visit_site = torch.tensor(
            indices,
            dtype=torch.long,
            device=self.y.device
        )
        # Init _mask_visits_each_site to None if not used
        self._mask_visits_each_site = None

        # DataLoaders for train/val/test splits.
        self.train_loader = None
        self.val_loader = None
        self.test_loader = None

    @property
    def mask_visits_each_site(self):
        """
        Visit indices grouped by site index.

        Returns
        -------
        list[torch.Tensor]
            ``i``-th tensor contains visit indices belonging to site ``i``.
        """
        if self._mask_visits_each_site is None:
            self._mask_visits_each_site = [
                torch.where(self.indices_visit_site == i)[0]
                for i in range(self.n_sites)
            ]
        return self._mask_visits_each_site

    def to(
        self, 
        *, 
        device: str | torch.device | None = None, 
        dtype: torch.dtype | None = None
    ) -> None:
        """
        Move internal tensors to a given device/dtype.
        Parameters
        ----------
        device : str or torch.device or None, default=None
            Device to move tensors to (e.g. "cpu", "cuda"). If None, no device move is performed.
        dtype : torch.dtype or None, default=None
            Dtype to move tensors to (e.g. torch.float32). If None, no dtype change is performed.
        Returns
        -------
        None
        """
        def _to(tensor: Optional[torch.Tensor]) -> Optional[torch.Tensor]:
            if tensor is None:
                return None
            if device is not None:
                tensor = tensor.to(device=device)
            if dtype is not None:
                tensor = tensor.to(dtype=dtype)
            return tensor
        self.y = _to(self.y)
        self.X_psi = _to(self.X_psi)
        self.X_d = _to(self.X_d)
        if device is not None: # indices_visit_site type should remain long, so only move to device if device is specified, but don't change dtype
            self.indices_visit_site = self.indices_visit_site.to(device=device)

    def build_dataloaders(
        self,
        *,
        batch_size: int = 32,
        val_fraction: float = 0.0,
        test_fraction: float = 0.0,
        splitting: Optional[Dict[str, list]] = None,
        shuffle: bool = True,
        seed: Optional[int] = None,
        device: str | torch.device | None = None,
    ):
        """
        Create train/validation/test dataloaders at site level.

        Parameters
        ----------
        batch_size : int, default=32
            Number of sites per batch.
        val_fraction : float, default=0.0
            Validation split fraction when ``splitting`` is not provided.
        test_fraction : float, default=0.0
            Test split fraction when ``splitting`` is not provided.
        splitting : dict[str, list] or None, default=None
            Explicit split with keys ``train``, ``val``, ``test``.
        shuffle : bool, default=True
            Whether to shuffle train batches.
        seed : int or None, default=None
            Random seed for split/shuffle.
        device : str or torch.device or None, default=None
            Optional device move before creating loaders.

        Returns
        -------
        None
        """
        if splitting is None:
            assert 0.0 <= val_fraction < 1.0, "val_fraction must be in [0.0, 1.0)"
            assert 0.0 <= test_fraction < 1.0, "test_fraction must be in [0.0, 1.0)"
            assert val_fraction + test_fraction < 1.0, "val_fraction + test_fraction must be < 1.0"

        if device is not None:
            self.to(device=device)
        dataset = OccuSiteDataset(self)
        n = len(dataset)

        if seed is not None:
            generator = torch.Generator().manual_seed(seed)
        else:
            generator = None

        if splitting is None:
            n_val = int(n * val_fraction)
            n_test = int(n * test_fraction)
            n_train = n - n_val - n_test
            train_ds, val_ds, test_ds = torch.utils.data.random_split(
                dataset, [n_train, n_val, n_test], generator=generator
            )
            # Get the ids corresponding to each split
            splitting = {
                "train": [dataset[i]["site_id"] for i in train_ds.indices],
                "val": [dataset[i]["site_id"] for i in val_ds.indices],
                "test": [dataset[i]["site_id"] for i in test_ds.indices],
            }
        else:
            assert set(splitting.keys()) == {"train", "val", "test"}, "splitting keys must be 'train', 'val', 'test'"
            all_idxs = set(range(len(dataset)))
            train_idxs = set(self.mapping_site_idx[id] for id in set(splitting["train"]))
            val_idxs = set(self.mapping_site_idx[id] for id in set(splitting["val"]))
            test_idxs = set(self.mapping_site_idx[id] for id in set(splitting["test"]))
            assert train_idxs.isdisjoint(val_idxs), "train and val indices must be disjoint"
            assert train_idxs.isdisjoint(test_idxs), "train and test indices must be disjoint"
            assert val_idxs.isdisjoint(test_idxs), "val and test indices must be disjoint"
            assert train_idxs.union(val_idxs).union(test_idxs) == all_idxs, "train, val, test indices must cover all samples"
            if train_idxs:
                train_ds = Subset(dataset, sorted(train_idxs))
                n_train = len(train_ds)
            else:
                train_ds, n_train = None, 0
            if val_idxs:
                val_ds = Subset(dataset, sorted(val_idxs))
                n_val = len(val_ds)
            else:
                val_ds, n_val = None, 0
            if test_idxs:
                test_ds = Subset(dataset, sorted(test_idxs))
                n_test = len(test_ds)
            else:
                test_ds, n_test = None, 0

        self.splitting = splitting

        self.train_loader = DataLoader(
            train_ds, batch_size=batch_size, shuffle=shuffle, 
            collate_fn=collate_occu_sites, generator=generator
        ) if n_train else None

        self.val_loader = DataLoader(
            val_ds, batch_size=batch_size, shuffle=False, 
            collate_fn=collate_occu_sites, generator=generator
        ) if n_val else None

        self.test_loader = DataLoader(
            test_ds, batch_size=batch_size, shuffle=False, 
            collate_fn=collate_occu_sites, generator=generator
        ) if n_test else None

    def reset_dataloaders(self):
        """
        Clear train/validation/test dataloaders.

        Returns
        -------
        None
        """
        self.train_loader = None
        self.val_loader = None
        self.test_loader = None

    def filtered_sites(self, site_id_mask: list) -> "OccuData":
        """
        Return a new ``OccuData`` restricted to selected site IDs.

        Parameters
        ----------
        site_id_mask : list
            Site IDs to keep.

        Returns
        -------
        OccuData
            Filtered dataset.
        """

        if len(site_id_mask) != len(set(site_id_mask)):
            warnings.warn(
                "site_mask contains duplicate indices. Duplicates are ignored.",
                UserWarning
            )
            site_id_mask = list(set(site_id_mask))

        indices_site_mask = [self.site_ids_in_X_psi.index(site_id) for site_id in site_id_mask]

        X_psi_new = self.X_psi[indices_site_mask] if self.X_psi is not None else None
        site_ids_in_X_psi_new = site_id_mask if self.X_psi is not None else None

        mapping_idxs_to_ids = {idx: site_id for site_id, idx in self.mapping_site_idx.items()}
        visit_site_ids = [mapping_idxs_to_ids[idx.item()] for idx in self.indices_visit_site]

        mask_v = torch.isin(self.indices_visit_site, torch.tensor(indices_site_mask))
        y_new = self.y[mask_v]
        visit_site_ids_new = [visit_site_ids[i] for i in torch.where(mask_v)[0].tolist()]
        X_d_new = self.X_d[mask_v] if self.X_d is not None else None

        return OccuData(
            y=y_new,
            visit_site_ids=visit_site_ids_new,
            X_psi=X_psi_new,
            X_d=X_d_new,
            site_ids_in_X_psi=site_ids_in_X_psi_new
        )

        *,
        df_visit: pd.DataFrame | pl.DataFrame, 
        df_site: pd.DataFrame | pl.DataFrame | None = None,
        col_visit_site_ids: str = "visit_site_id",
        col_site_ids_in_X_psi: str | None = "site_id",
        cols_y: list[str] | None = None,
        cols_covars_d: list[str] | None = None,
        cols_covars_psi: list[str] | None = None,
        device: torch.device,
        verbose: bool = False
) -> dict:
    """
    Build an ``OccuData`` constructor payload from visit/site dataframes.

    Parameters
    ----------
    df_visit : pandas.DataFrame or polars.DataFrame
        Visit-level table (detections and optional visit covariates).
    df_site : pandas.DataFrame or polars.DataFrame or None, default=None
        Site-level table (optional occupancy covariates).
    col_visit_site_ids : str, default="visit_site_id"
        Site-id column in ``df_visit``.
    col_site_ids_in_X_psi : str or None, default="site_id"
        Site-id column in ``df_site``.
    cols_y : list[str] or None, default=None
        Detection columns.
    cols_covars_d : list[str] or None, default=None
        Detection covariate columns.
    cols_covars_psi : list[str] or None, default=None
        Occupancy covariate columns.
    device : torch.device
        Output tensor device.
    verbose : bool, default=False
        If ``True``, prints shape summary.

    Returns
    -------
    dict
        Dictionary with keys ``y``, ``visit_site_ids``, ``site_ids_in_X_psi``,
        ``X_psi``, and ``X_d``.
    """
    if not isinstance(df_visit, (pd.DataFrame, pl.DataFrame)):
        raise ValueError("df_visit must be a pandas or polars DataFrame")
    if df_site is not None and not isinstance(df_site, (pd.DataFrame, pl.DataFrame)):
        raise ValueError("df_site must be a pandas or polars DataFrame or None")

    # Define y
    if isinstance(df_visit, pd.DataFrame):
        y = torch.tensor(df_visit[cols_y].to_numpy(), dtype=torch.float32, device=device) # shape: (n_visits, n_species)
    elif isinstance(df_visit, pl.DataFrame):
        y = torch.tensor(df_visit.select(cols_y).to_numpy(), dtype=torch.float32, device=device) # shape: (n_visits, n_species)

    # Define visit_site_ids
    if isinstance(df_visit, pd.DataFrame):
        visit_site_ids = df_visit[col_visit_site_ids].to_list() # length: n_visits
    elif isinstance(df_visit, pl.DataFrame):
        visit_site_ids = df_visit.select(col_visit_site_ids).to_series().to_list() # length: n_visits

    # Define site_ids_in_X_psi (could be None) and check consistency with visit_site_ids
    if df_site is None:
        site_ids_in_X_psi = None
    elif col_site_ids_in_X_psi is None:
        if cols_covars_psi is None:
            site_ids_in_X_psi = None
        else:

    else:
        if isinstance(df_site, pd.DataFrame):
            site_ids_in_X_psi = df_site[col_site_ids_in_X_psi].to_list() # length: n_sites_in_X_psi
        elif isinstance(df_site, pl.DataFrame):

    # Define X_psi
    if cols_covars_psi is None or len(cols_covars_psi) == 0:
        X_psi = None
    else:
        if isinstance(df_site, pd.DataFrame):
            X_psi = torch.tensor(df_site[cols_covars_psi].to_numpy(), dtype=torch.float32, device=device) # shape: (n_sites_in_X_psi, n_cov_psi)
        elif isinstance(df_site, pl.DataFrame):
            X_psi = torch.tensor(df_site.select(cols_covars_psi).to_numpy(), dtype=torch.float32, device=device) # shape: (n_sites_in_X_psi, n_cov_psi)

    # Define X_d
    if cols_covars_d is None or len(cols_covars_d) == 0:
        X_d = None
    else:
        if isinstance(df_visit, pd.DataFrame):
            X_d = torch.tensor(df_visit[cols_covars_d].to_numpy(), dtype=torch.float32, device=device) # shape: (n_visits, n_cov_d)
        elif isinstance(df_visit, pl.DataFrame):
            X_d = torch.tensor(df_visit.select(cols_covars_d).to_numpy(), dtype=torch.float32, device=device) # shape: (n_visits, n_cov_d)

    # Print shapes and lengths if needed
    if verbose:
        print(f"y shape (n_visits, n_species): {y.shape}")
        print(f"visit_site_ids length (n_visits): {len(visit_site_ids)}")
        print(f"site_ids_in_X_psi length (n_sites): {len(site_ids_in_X_psi)}")
        if X_psi is not None:
            print(f"X_psi shape (n_sites, n_cov_psi): {X_psi.shape}")

        if X_d is not None:
            print(f"X_d shape (n_visits, n_cov_d): {X_d.shape}")

        "X_d": X_d

    def simulate_linear(
        n_sites=400, n_total_visits_max=None,
        f_effort=lambda x1: torch.exp(1.5*x1), 
        keep_all_sites=False,
        intercept_psi=torch.tensor([0.0]), # shape: (n_species,)
        coefs_psi=torch.tensor([[0.0]]), # shape: (n_species, n_cov_psi)
        intercept_d=torch.tensor([0.0]), # shape: (n_species,)
        coefs_d=torch.tensor([[0.0]]), # shape: (n_species, n_cov_d)
        seed=None, 
        verbose=True, 
        plot_summary=True
    ):
    """
    Simulate occupancy data from linear logits for psi and d.

    Returns
    -------
    tuple
        ``(y, visit_site_ids, site_ids_in_X_psi, X_psi, X_d, z)``.
    """
    g = torch.Generator().manual_seed(seed) if seed is not None else None

    X_psi = torch.randn(n_sites, len(coefs_psi[0]), generator=g)  # shape: (n_sites, n_cov_psi)

    # Define the number of visits per site
    n_visits_per_site = f_effort(X_psi[:, 0]).round().long()  # shape: (n_sites,)
    n_visits_per_site = n_visits_per_site.clamp(min=int(keep_all_sites))  # shape: (n_sites,) (if keep_all_sites is True, ensure at least 1 visit per site)
    # fig, axes = plt.subplots(1, 2, figsize=(8, 3))

        print(f"  observed average psi (knowing true occupancy): {z.mean(dim=0)}")

    def simulate_non_linear(
        n_sites=400, 
        n_total_visits_max=None,
        n_cov_psi=2,
        n_cov_d=1,
        f_effort=lambda x1: torch.exp(1.5*x1), 
        keep_all_sites=False,
        f_psi=lambda X_psi: torch.sigmoid(
            torch.tensor([-1.0, 0.5]) + X_psi @ torch.tensor([[2.0, -1.0], [-1.0, 1.0]]).T
        ), # from X_psi of shape (n_sites, n_cov_psi) to psi of shape (n_sites, n_species)
        f_d=lambda X_d: torch.sigmoid(
            torch.tensor([-0.5, 1.0]) + X_d @ torch.tensor([[1.0], [-2.0]]).T
        ), # from X_d of shape (n_visits, n_cov_d) to d of shape (n_visits, n_species)
        seed=None,
        plot_summary=True,
        verbose=True
    ):
    """
    Simulate occupancy data using user-defined nonlinear ``f_psi`` and ``f_d``.

    Returns
    -------
    tuple
        ``(y, visit_site_ids, site_ids_in_X_psi, X_psi, X_d, z)``.
    """
    g = torch.Generator().manual_seed(seed) if seed is not None else None

    X_psi = torch.randn(n_sites, n_cov_psi, generator=g)  # shape: (n_sites, n_cov_psi)

    # Define the number of visits per site
    n_visits_per_site = f_effort(X_psi[:, 0]).round().long()  # shape: (n_sites,)
    n_visits_per_site = n_visits_per_site.clamp(min=int(keep_all_sites))  # shape: (n_sites,) (if keep_all_sites is True, ensure at least 1 visit per site)
    indices_visit_site = torch.arange(n_sites).repeat_interleave(n_visits_per_site)  # shape: (n_visits,)
    # Filter out sites with zero visits
    if n_total_visits_max is not None: # Randomly subsample visits to have at most n_total_visits_max
        if indices_visit_site.shape[0] > n_total_visits_max:
            indices_visit_site = filter_subset_visits(
                indices_visit_site, n_sites, n_total_visits_max, 
                generator=g, keep_all_sites=keep_all_sites,
                verbose=verbose
            )
    n_visits_per_site = torch.bincount(indices_visit_site, minlength=n_sites)  # shape: (n_sites,) (for each site, how many visits it has)
    # Reassign site indices to be consecutive after filtering
    map_sites_to_sub = {}
    curr_idx = 0
    for i in range(n_sites):
        if n_visits_per_site[i] > 0:
            map_sites_to_sub[i] = curr_idx
            curr_idx += 1
    n_sites_sub = curr_idx
    mapping = torch.full((n_sites,), -1, dtype=torch.long, device=indices_visit_site.device)
    keys = torch.tensor(list(map_sites_to_sub.keys()), device=mapping.device)
    values = torch.tensor(list(map_sites_to_sub.values()), device=mapping.device)
    mapping[keys] = values
    indices_visit_site = mapping[indices_visit_site]
    n_visits_sub = indices_visit_site.shape[0]
    X_psi = X_psi[keys]

    X_d = torch.randn(n_visits_sub, n_cov_d, generator=g)  # shape: (n_visits_sub, n_cov_d)

    psi = f_psi(X_psi)  # shape: (n_sites_sub, n_species)
    z = torch.bernoulli(psi, generator=g)  # shape: (n_sites_sub, n_species)
    z_for_visit = z[indices_visit_site]  # shape: (n_visits_sub, n_species)

    d = f_d(X_d)  # shape: (n_visits_sub, n_species)

    y = torch.bernoulli(d * z_for_visit, generator=g)  # shape: (n_visits_sub, n_species)

    # Plot summary of simulated data
    if verbose:
        detected = torch.zeros((n_sites, y.shape[1]), device=y.device, dtype=y.dtype)
        detected.index_add_(0, indices_visit_site, y)
        detected = (detected > 0) # shape: (n_sites, n_species)
        print("--- Simulated data summary ---")
        print(f"  n_sites={n_sites_sub}")
        print(f"  n_visits={n_visits_sub}")
        print(f"  number of detections (per species): {y.sum(dim=0)}")
        print(f"  number of occupied sites (per species): {detected.sum(dim=0)}")
        print(f"  range of psi * d: {(psi[indices_visit_site] * d).min():3e} to { (psi[indices_visit_site] * d).max():3e}")
        print(f"  observed average psi (knowing true occupancy): {z.mean(dim=0)}")
        print(f"  observed average d (knowing true occupancy):   {y.sum(dim=0) / z_for_visit.sum(dim=0)}")

    if plot_summary:
        fig, axes = plt.subplots(psi.shape[1]+1, 3, figsize=(10, (psi.shape[1]+1)*2))
        gs = fig.add_gridspec(psi.shape[1]+1, 3)
        # Histogram of n_visits_per_site
        ax0 = fig.add_subplot(gs[0, :])
        logbins = torch.logspace(torch.log10(n_visits_per_site[n_visits_per_site>0].min()), torch.log10(n_visits_per_site.max()), 30)
        ax0.hist(n_visits_per_site.cpu(), bins=logbins, edgecolor='k')
        ax0.set_xscale("log")
        # ax0.hist(n_visits_per_site.cpu(), bins=50, edgecolor='k')
        ax0.set_xlabel("Number of visits per site")
        ax0.set_ylabel("Frequency")
        ax0.set_title(f"Distribution of visits per site ({(n_visits_per_site>0).sum()}/{n_sites} sites with >0 visits)")
        ax0.grid(True)
        ax0.set_yscale("log")
        for i in range(3):
            axes[0, i].axis('off')
        # Histogram of psi
        for species_idx in range(psi.shape[1]):
            tmp = psi[:, species_idx]
            logbins = torch.logspace(torch.log10(tmp.min()), torch.log10(tmp.max()), 30)
            axes[species_idx+1, 0].hist(tmp.cpu(), bins=logbins, edgecolor='k')
            axes[species_idx+1, 0].set_xlabel("Occupancy probability (psi)")
            axes[species_idx+1, 0].set_ylabel("Frequency")
            axes[species_idx+1, 0].grid(True)
            axes[species_idx+1, 0].set_title(f"psi (species {species_idx})")
            axes[species_idx+1, 0].set_xscale("log")
        # Histogram of d
        for species_idx in range(d.shape[1]):
            tmp = d[:, species_idx]
            logbins = torch.logspace(torch.log10(tmp.min()), torch.log10(tmp.max()), 30)
            axes[species_idx+1, 1].hist(tmp.cpu(), bins=logbins, edgecolor='k')
            axes[species_idx+1, 1].set_xlabel("Detection probability (d)")
            axes[species_idx+1, 1].set_ylabel("Frequency")
            axes[species_idx+1, 1].grid(True)
            axes[species_idx+1, 1].set_title(f"d (species {species_idx})")
            axes[species_idx+1, 1].set_xscale("log")
        # Histogram of psi*d
        for species_idx in range(psi.shape[1]):
            tmp = psi[indices_visit_site, species_idx] * d[:, species_idx]
            logbins = torch.logspace(torch.log10(tmp.min()), torch.log10(tmp.max()), 30)
            axes[species_idx+1, 2].hist(tmp.cpu(), bins=logbins, edgecolor='k')
            axes[species_idx+1, 2].set_xlabel("Effective detection probability (psi * d)")
            axes[species_idx+1, 2].set_ylabel("Frequency")
            axes[species_idx+1, 2].grid(True)
            axes[species_idx+1, 2].set_title(f"psi*d (species {species_idx})")
            axes[species_idx+1, 2].set_xscale("log")
        fig.tight_layout()

    visit_site_ids = indices_visit_site.long().tolist()
    # unique site ids, preserving the order of first occurence in visit_site_ids
    site_ids_in_X_psi = list(dict.fromkeys(visit_site_ids))

    # Filter X_psi on sites present in visit_site_ids
    X_psi = X_psi[site_ids_in_X_psi]

    return y.float(), visit_site_ids, site_ids_in_X_psi, X_psi.float(), X_d.float(), z.float()

    indices_visit_site = data.indices_visit_site

    if indices_visit_site.numel() == 0:
        raise ValueError("Cannot fit with zero visits.")

    X_psi = data.X_psi
    X_d = data.X_d

    # Penalization setup
    if penalization_type in ["l2_logits_naive", "l1_logits_naive"]:
        from .naive_estimator import fit_naive_estimator
        center_logit_psi, center_logit_d, _, _ = fit_naive_estimator(
            data=data,
            loss_fn=loss_fn,
            device=device
        ) # naive logits of shape: (n_species,)
    elif penalization_type in ["l2_logits", "l1_logits"]:
        center_logit_psi = torch.zeros((model.n_species,), device=device)
        center_logit_d = torch.zeros((model.n_species,), device=device)
    else:
        center_logit_psi, center_logit_d = None, None

    # LBFGS optimizer setup
    lbfgs_kwargs = dict(max_iter=max_iter)

    if tolerance_grad is not None:
        lbfgs_kwargs["tolerance_grad"] = tolerance_grad
    if tolerance_change is not None:
        lbfgs_kwargs["tolerance_change"] = tolerance_change
    if history_size is not None:
        lbfgs_kwargs["history_size"] = history_size
    if line_search_fn is not None:
        lbfgs_kwargs["line_search_fn"] = line_search_fn


    parameters = list(model.psi_model.parameters()) + list(model.d_model.parameters())
    parameters = [p for p in parameters if p.requires_grad]
    if len(parameters) == 0:
        raise ValueError("No parameters to optimize. Ensure that model parameters have requires_grad=True.")
    opt = torch.optim.LBFGS(parameters, lr=lr, **lbfgs_kwargs)

    # Closure function for LBFGS optimization
    n_iter = 0
    last_loss = None
    train_loss_history = []
    eval_loss_history = []
    pbar = None

    def closure():

        nonlocal n_iter, last_loss, train_loss_history, eval_loss_history, \
            pbar, path_checkpoints, model, opt, lr


        if n_iter == 0:
            pbar = tqdm(total=max_iter, desc="LBFGS iterations", unit="iter")
        opt.zero_grad(set_to_none=True)

        logit_psi, logit_d = model.forward_logits(X_psi=X_psi, X_d=X_d)

        # Expand logits if X_psi or X_d were None (model would output shape (1, n_species) 
        # but we need to broadcast to all sites/visits for loss calculation)
        if X_psi is None:
            logit_psi = logit_psi.expand(data.n_sites, -1)
        if X_d is None:
            logit_d = logit_d.expand(data.n_visits, -1)

        # Basic shape validation (helps catch mistakes in custom models early)
        if logit_psi.ndim != 2 or logit_d.ndim != 2:
            raise ValueError("Model must return 2D tensors for (logit_psi, logit_d).")
        if logit_psi.shape[0] != data.n_sites:
            raise ValueError(
                f"Model returned logit_psi with n_sites={logit_psi.shape[0]}, expected {data.n_sites}."
            )
        if logit_d.shape[0] != data.n_visits:
            raise ValueError(
                f"Model returned logit_d with n_visits={logit_d.shape[0]}, expected {data.n_visits}."
            )
        if logit_psi.shape[1] != y.shape[1] or logit_d.shape[1] != y.shape[1]:
            raise ValueError("Model species dimension must match y.shape[1].")

        nll = loss_fn(
            y,
            logit_psi=logit_psi,
            logit_d=logit_d,
            indices_visit_site=indices_visit_site,
            site_reduction=site_reduction,
            species_reduction=species_reduction,
            validate_inputs=n_iter==0 # only validate on first iteration for performance
        )

        penalty = loss_penalty(
            penalization_type=penalization_type,
            penalization_lambda_psi=penalization_lambda_psi,
            penalization_lambda_d=penalization_lambda_d,
            model=model,
            logit_psi=logit_psi,
            logit_d=logit_d,
            center_logit_psi=center_logit_psi,
            center_logit_d=center_logit_d,
            site_reduction=site_reduction,
            species_reduction=species_reduction
        )
        loss = nll + penalty

        if not torch.isfinite(loss):
            raise ValueError(f"Non-finite loss encountered: {loss.item()}")
        loss.backward()
        last_loss = loss.detach()
        n_iter += 1
        train_loss_history.append(float(last_loss.cpu().item()))

        if eval_loss:
            this_eval_loss = loss_fn_eval(
                y,
                logit_psi=logit_psi,
                logit_d=logit_d,
                indices_visit_site=indices_visit_site,
                site_reduction=site_reduction,
                species_reduction=species_reduction,
            ).detach()
            eval_loss_history.append(float(this_eval_loss.cpu().item()))

        pbar.set_postfix({"train_loss": last_loss.cpu().item()})
        pbar.update(1)

        # Save the checkpoint
        save_checkpoint_if_best(
            path_checkpoints=path_checkpoints,
            epoch=n_iter-1,
            last_train_loss=last_loss,
            last_val_loss=None,
            final_params_model_psi=deepcopy(model.psi_model.state_dict()),
            final_params_model_d=deepcopy(model.d_model.state_dict()),
            opt_psi=None,
            opt_d=None,
            opt=opt,
            optimizer_psi_cls=torch.optim.LBFGS,
            optimizer_d_cls=torch.optim.LBFGS,
            train_loss_history=train_loss_history,
            val_loss_history=eval_loss_history,
            lr_psi_history=[lr] * n_iter,
            lr_d_history=[lr] * n_iter,
            verbose=False
        )

        return loss

    # Fit the model using LBFGS optimization
    print(f"Starting LBFGS fitting on device {device}")
    loss = opt.step(closure)
    pbar.close()

    print(f"LBFGS finished after {n_iter} iterations with final train loss {last_loss.cpu().item():.4f}")

    final = last_loss if last_loss is not None else loss.detach()

    save_checkpoint_if_best(
        path_checkpoints=path_checkpoints,
        # splitting=data.splitting,
        epoch=n_iter-1,
        last_train_loss=last_loss,
        last_val_loss=None,
        final_params_model_psi=deepcopy(model.psi_model.state_dict()),
        final_params_model_d=deepcopy(model.d_model.state_dict()),
        opt_psi=None,
        opt_d=None,
        opt=opt,
        optimizer_psi_cls=torch.optim.LBFGS,
        optimizer_d_cls=torch.optim.LBFGS,
        train_loss_history=train_loss_history,
        val_loss_history=eval_loss_history,
        lr_psi_history=[lr] * n_iter,
        lr_d_history=[lr] * n_iter,
        verbose=False
    )

    # LBFGS doesn't expose a robust convergence flag; keep best-effort.
    return FitResult(
        final_train_loss=float(final.cpu().item()), 
        final_val_loss=None,
        n_epochs=n_iter, 
        converged=True,
        history={"train_loss": train_loss_history, "val_loss": eval_loss_history},
        lr_psi=lr,
        lr_d=lr,
        optimizer_psi="LBFGS",
        optimizer_d="LBFGS",
        device=device
    )

    def fit_sgd_like(
    data: OccuData,
    model: OccuModel,
    *,
    batch_size: int = 256,
    n_epochs: int = 50,
    lr_psi: float = 1e-3,
    lr_d: float = 1e-3,
    optimizer_psi_cls: Callable = torch.optim.Adam,
    optimizer_d_cls: Callable = torch.optim.Adam,
    site_reduction: str = "sum",
    species_reduction: str = "mean",
    loss_fn: Callable[..., torch.Tensor] = nll_logits,
    loss_fn_eval: Optional[Callable[..., torch.Tensor]] = None,
    device: Optional[torch.device] = None,
    seed_dataloader: Optional[int] = None,
    val_fraction: float = 0.0,
    test_fraction: float = 0.0,
    splitting: Optional[Dict[str, list]] = None,
    patience_early_stopping: Optional[int] = None,
    min_delta_early_stopping: float = 0.0,
    score_early_stopping: str = "val_loss",
    patience_reduceLROnPlateau: Optional[int] = None,
    factor_reduceLROnPlateau: float = 0.5,
    min_lr_reduceLROnPlateau: float = 1e-5,
    score_reduceLROnPlateau: str = "val_loss",
    penalization_type: Optional[str] = None,
    penalization_lambda_psi: Optional[float] = None,
    penalization_lambda_d: Optional[float] = None,
    path_checkpoints: Optional[str] = None
) -> FitResult:
    """
    Fit an occupancy model with mini-batch SGD-like optimization.

    Parameters
    ----------
    data : OccuData
        Input data.
    model : OccuModel
        Model to optimize (modified in place).
    batch_size : int, default=256
        Number of sites per batch.
    n_epochs : int, default=50
        Maximum number of epochs.
    lr_psi : float, default=1e-3
        Learning rate for occupancy parameters.
    lr_d : float, default=1e-3
        Learning rate for detection parameters.
    optimizer_psi_cls, optimizer_d_cls : callable, default=torch.optim.Adam
        Optimizer classes for occupancy and detection branches.
    site_reduction : str, default="sum"
        Reduction over sites for the loss ("sum", "mean", or "none").
    species_reduction : str, default="mean"
        Reduction over species for the loss ("sum", "mean", or "none").
    loss_fn : callable, default=nll_logits
        Loss function to optimize.
    loss_fn_eval : callable or None, default=None
        Loss function for validation. If None, uses loss_fn.
    device : torch.device or None, default=None
        Device for computation. If None, uses the device of the model parameters.
    seed_dataloader : int or None, default=None
        Random seed for dataloader shuffling.
    val_fraction : float, default=0.0
        Fraction of data to use for validation if no val_loader is provided.
    test_fraction : float, default=0.0
        Fraction of data to use for testing if no test_loader is provided.
    splitting : dict or None, default=None
        Explicit splitting of sites into "train", "val", and "test" lists. Overrides val_fraction and test_fraction if provided.
    patience_early_stopping : int or None, default=None
        Number of epochs with no improvement after which training will be stopped. If None, no early stopping is performed.
    min_delta_early_stopping : float, default=0.0
        Minimum change in the monitored score to qualify as an improvement for early stopping.
    score_early_stopping : str, default="val_loss"
        Score to monitor for early stopping ("val_loss" or "train_loss").
    patience_reduceLROnPlateau : int or None, default=None
        Number of epochs with no improvement after which learning rate will be reduced. If None, no learning rate scheduling is performed.
    factor_reduceLROnPlateau : float, default=0.5
        Factor by which the learning rate will be reduced when using ReduceLROnPlateau.
    min_lr_reduceLROnPlateau : float, default=1e-5
        Minimum learning rate to which the optimizer will be reduced.
    score_reduceLROnPlateau : str, default="val_loss"
        Score to monitor for ReduceLROnPlateau ("val_loss" or "train_loss").
    penalization_type : str or None, default=None
        Type of penalization to apply ("l2_params", "l1_params", "l2_logits", "l1_logits", "l2_logits_naive", "l1_logits_naive", or None).
    penalization_lambda_psi : float or None, default=None
        Strength of penalization for occupancy parameters (used if penalization_type is "l2_params" or "l1_params").
    penalization_lambda_d : float or None, default=None
        Strength of penalization for detection parameters (used if penalization_type is "l2_params" or "l1_params").
    path_checkpoints : str or None, default=None
        Path where best checkpoint is saved.

    Returns
    -------
    FitResult
        Training summary and history.

    Notes
    -----
    Supported penalties:
    ``None, l2_params, l1_params, l2_logits, l1_logits, l2_logits_naive, l1_logits_naive``.
    The model is modified in-place. 
    If using early stopping, the best model is loaded at the end. 
    Otherwise, the final model is returned.
    """

    if device is None:
        device = next(model.psi_model.parameters()).device

    if loss_fn_eval is None:
        loss_fn_eval = loss_fn

    # Build dataloader (sites batched)
    if data.train_loader is None:
        print("Building dataloader for training...")
        data.build_dataloaders(
            batch_size=batch_size,
            val_fraction=val_fraction,
            test_fraction=test_fraction,
            splitting=splitting,
            shuffle=True,
            seed=seed_dataloader,
            device=device
        )
    else:
        n_sites_in_dataloader = len(set(next(iter(data.train_loader))["indices_visit_site"]))
        if n_sites_in_dataloader != batch_size:
            warnings.warn(
                f"Provided dataloader has batch size {n_sites_in_dataloader}, "
                f"which does not match the specified batch_size {batch_size}. "
                f"Using the provided dataloader's batch size.",
                UserWarning
            )

    # Penalization setup
    if penalization_type in ["l2_logits_naive", "l1_logits_naive"]:
        if splitting is None or "train" not in splitting:
            if data.train_loader is None:
                raise ValueError("Splitting information is required for naive logits penalization. Please provide a splitting dictionary with a 'train' key or ensure that the dataloader is built with a train split.")
            else:
                train_sites_indices = []
                for batch in data.train_loader:
                    batch_site_indices = batch["indices_visit_site"].cpu().numpy()
                    train_sites_indices.extend(batch_site_indices)
                train_sites_indices = sorted(set(int(x) for x in train_sites_indices))
                train_sites_ids = [data.site_ids_in_X_psi[i] for i in train_sites_indices]
        else:
            train_sites_ids = splitting["train"]
        center_logit_psi, center_logit_d, _, _ = fit_naive_estimator(
            data=data.filtered_sites(train_sites_ids),
            loss_fn=loss_fn,
            device=device
        ) # naive logits of shape: (n_species,)
    elif penalization_type in ["l2_logits", "l1_logits"]:
        center_logit_psi = torch.zeros((model.n_species,), device=device)
        center_logit_d = torch.zeros((model.n_species,), device=device)
    else:
        center_logit_psi, center_logit_d = None, None

    # Optimizer
    parameters_psi = [p for p in model.psi_model.parameters() if p.requires_grad]
    parameters_d = [p for p in model.d_model.parameters() if p.requires_grad]
    # if sum(p.numel() for p in parameters_psi) + sum(p.numel() for p in parameters_d) == 0:
    if len(parameters_psi) + len(parameters_d) == 0:
        raise ValueError("No parameters to optimize. Ensure that model parameters have requires_grad=True.")
    if optimizer_psi_cls == torch.optim.LBFGS or optimizer_d_cls == torch.optim.LBFGS:
        if optimizer_psi_cls != optimizer_d_cls:
            raise ValueError("LBFGS optimizer must be used for both psi and d.")
        if lr_psi != lr_d:
            raise ValueError("LBFGS optimizer must use the same learning rate for psi and d.")
        opt = torch.optim.LBFGS(parameters_psi + parameters_d, lr=lr_psi, max_iter=1)
        opt_psi, opt_d = None, None
    else:
        # NB: L2 regularization is implemented as weight decay in the optimizer
        opt_psi = optimizer_psi_cls(
            parameters_psi,
            lr=lr_psi, 
            weight_decay=penalization_lambda_psi if penalization_type == "l2_params" else 0.0
        )
        opt_d = optimizer_d_cls(
            parameters_d, 
            lr=lr_d, 
            weight_decay=penalization_lambda_d if penalization_type == "l2_params" else 0.0
        )
        opt = None

    # Early stopping
    if patience_early_stopping is not None:
        if data.val_loader is None and score_early_stopping == "val_loss":
            raise ValueError("Early stopping requires a validation set. Please provide a val_loader in the data or set patience_early_stopping to None.")
        elif score_early_stopping not in ["val_loss", "train_loss"]:
            raise ValueError(f"Invalid score_early_stopping: {score_early_stopping}. Must be 'val_loss' or 'train_loss'.")
        early_stopper = EarlyStopper(
            patience=patience_early_stopping, 
            min_delta=min_delta_early_stopping
        )

    # LR scheduler
    if patience_reduceLROnPlateau is not None:
        if data.val_loader is None and score_reduceLROnPlateau == "val_loss":
            raise ValueError("ReduceLROnPlateau requires a validation set if score_reduceLROnPlateau is set to 'val_loss'. Please provide a val_loader in the data or set patience_reduceLROnPlateau to None or score_reduceLROnPlateau to 'train_loss'.")
        elif score_reduceLROnPlateau not in ["val_loss", "train_loss"]:
            raise ValueError(f"Invalid score_reduceLROnPlateau: {score_reduceLROnPlateau}. Must be 'val_loss' or 'train_loss'.")
        from torch.optim.lr_scheduler import ReduceLROnPlateau
        kwargs_scheduler = {
            "mode": "min",
            "factor": factor_reduceLROnPlateau,
            "patience": patience_reduceLROnPlateau,
            "min_lr": min_lr_reduceLROnPlateau
        }
        if optimizer_psi_cls == torch.optim.LBFGS:
            scheduler = ReduceLROnPlateau(opt, **kwargs_scheduler)
        else:
            scheduler_state = ReduceLROnPlateau(opt_psi, **kwargs_scheduler)
            scheduler_det = ReduceLROnPlateau(opt_d, **kwargs_scheduler)

    # Training loop
    train_loss_history = []
    val_loss_history = []
    lr_psi_history, lr_d_history = [], []

    data.to(device=device)
    model.to(device=device)
    print(f"Starting SGD-like fitting on device {device}")

    final_params_model_psi = deepcopy(model.psi_model.state_dict())
    final_params_model_d = deepcopy(model.d_model.state_dict())
    final_train_loss, final_val_loss = float('inf'), float('inf')
    epoch_best_model = 0

    pbar = tqdm(range(n_epochs), desc="Epochs", unit="epoch")
    for epoch in range(n_epochs):

        # Training steps
        model.train()
        last_train_loss = 0.0
        for batch in data.train_loader:

            if optimizer_psi_cls == torch.optim.LBFGS:
                def closure():
                    opt.zero_grad()
                    nll, logit_psi, logit_d = forward_path_one_batch(
                        batch=batch,
                        model=model,
                        loss_fn=loss_fn,
                        site_reduction=site_reduction,
                        species_reduction=species_reduction,
                        device=device,
                        validate_inputs_loss_fn=epoch == 0 # validate inputs only in the first epoch for performance
                    )
                    penalty = loss_penalty(
                        penalization_type=penalization_type,
                        penalization_lambda_psi=penalization_lambda_psi,
                        penalization_lambda_d=penalization_lambda_d,
                        model=model,
                        logit_psi=logit_psi,
                        logit_d=logit_d,
                        center_logit_psi=center_logit_psi,
                        center_logit_d=center_logit_d,
                        site_reduction=site_reduction,
                        species_reduction=species_reduction
                    )
                    loss = nll + penalty
                    loss.backward()
                    return loss
                opt.step(closure)
                last_train_loss += closure().cpu().item()

            else:
                opt_psi.zero_grad()
                opt_d.zero_grad()
                nll, logit_psi, logit_d = forward_path_one_batch(
                    batch=batch,
                    model=model,
                    loss_fn=loss_fn,
                    site_reduction=site_reduction,
                    species_reduction=species_reduction,
                    device=device,
                    validate_inputs_loss_fn=epoch == 0 # validate inputs only in the first epoch for performance
                )
                penalty = loss_penalty(
                    penalization_type=penalization_type if penalization_type != "l2_params" else None, # L2 on parameters is handled by weight decay in the optimizer
                    penalization_lambda_psi=penalization_lambda_psi,
                    penalization_lambda_d=penalization_lambda_d,
                    model=model,
                    logit_psi=logit_psi,
                    logit_d=logit_d,
                    center_logit_psi=center_logit_psi,
                    center_logit_d=center_logit_d,
                    site_reduction=site_reduction,
                    species_reduction=species_reduction
                )
                loss = nll + penalty
                loss.backward()
                opt_psi.step()
                opt_d.step()
                last_train_loss += loss.detach()

        train_loss_history.append(float(last_train_loss.cpu().item()))

        # Validation steps
        if data.val_loader is not None:
            model.eval()
            last_val_loss = 0.0
            with torch.no_grad():
                for batch in data.val_loader:
                    val_loss, _, _ = forward_path_one_batch(
                        batch=batch,
                        model=model,
                        loss_fn=loss_fn_eval,
                        site_reduction=site_reduction,
                        species_reduction=species_reduction,
                        device=device,
                        validate_inputs_loss_fn=epoch == 0 # validate inputs only in the first epoch for performance
                    )
                    last_val_loss += val_loss.detach()
            val_loss_history.append(float(last_val_loss.cpu().item()))
        else:
            last_val_loss = None

        lr_psi_history.append(opt_psi.param_groups[0]['lr'] if optimizer_psi_cls != torch.optim.LBFGS else opt.param_groups[0]['lr'])
        lr_d_history.append(opt_d.param_groups[0]['lr'] if optimizer_d_cls != torch.optim.LBFGS else opt.param_groups[0]['lr'])

        # Store best (eventually final) model and losses
        if data.val_loader is not None:
            if last_val_loss < final_val_loss:
                final_train_loss = last_train_loss
                final_val_loss = last_val_loss
                final_params_model_psi = deepcopy(model.psi_model.state_dict())
                final_params_model_d = deepcopy(model.d_model.state_dict())
                epoch_best_model = epoch
        else:
            if last_train_loss < final_train_loss:
                final_train_loss = last_train_loss
                final_params_model_psi = deepcopy(model.psi_model.state_dict())
                final_params_model_d = deepcopy(model.d_model.state_dict())
                epoch_best_model = epoch

        save_checkpoint_if_best(
            path_checkpoints=path_checkpoints,
            epoch=epoch,
            last_train_loss=last_train_loss,
            last_val_loss=last_val_loss,
            final_params_model_psi=final_params_model_psi,
            final_params_model_d=final_params_model_d,
            opt_psi=opt_psi,
            opt_d=opt_d,
            opt=opt,
            optimizer_psi_cls=optimizer_psi_cls,
            optimizer_d_cls=optimizer_d_cls,
            train_loss_history=train_loss_history,
            val_loss_history=val_loss_history,
            lr_psi_history=lr_psi_history,
            lr_d_history=lr_d_history,
            verbose=False
        )

        # Update progress bar
        pbar.set_postfix({
            "train_loss": last_train_loss,
            "val_loss": last_val_loss if data.val_loader is not None else "N/A"
        })
        pbar.update(1)

        # Early stopping check
        if patience_early_stopping is not None:
            if score_early_stopping == "val_loss":
                score_ES = last_val_loss
            elif score_early_stopping == "train_loss":
                score_ES = last_train_loss
            if early_stopper.early_stop(score_ES):
                print(f"Early stopping at epoch n°{epoch + 1}")
                break

        # LR scheduler step
        if patience_reduceLROnPlateau is not None:
            if score_reduceLROnPlateau == "val_loss":
                score_RLROP = last_val_loss
            elif score_reduceLROnPlateau == "train_loss":
                score_RLROP = last_train_loss
            if optimizer_psi_cls == torch.optim.LBFGS:
                scheduler.step(score_RLROP)
            else:
                scheduler_state.step(score_RLROP)
                scheduler_det.step(score_RLROP)

    pbar.close()

    history = {"train_loss": train_loss_history, "lr_psi": lr_psi_history, "lr_d": lr_d_history}
    if len(val_loss_history) > 0:
        history["val_loss"] = val_loss_history

    final_val_loss = last_val_loss if data.val_loader is not None else None
    model.psi_model.load_state_dict(final_params_model_psi)
    model.d_model.load_state_dict(final_params_model_d)

    print(f"SGD-like fitting finished after {n_epochs} epochs.\n"
          f"Best model at epoch n°{epoch_best_model + 1} with final train loss {final_train_loss:.4f}" + 
          (f" and final val loss {final_val_loss:.4f}" if data.val_loader is not None else ""))

    return FitResult(
        final_train_loss=final_train_loss,
        final_val_loss=final_val_loss,
        n_epochs=epoch + 1,
        idx_epoch_best_model=epoch_best_model,
        converged=True,
        history=history,
        lr_psi=lr_psi,
        optimizer_psi=optimizer_psi_cls.__name__,
        lr_d=lr_d,
        optimizer_d=optimizer_d_cls.__name__,
        device=device
    )

    def loss_penalty(
        penalization_type: Optional[str],
        penalization_lambda_psi: Optional[float],
        penalization_lambda_d: Optional[float],
        model: OccuModel,
        logit_psi: Optional[torch.Tensor], # shape: (n_sites, n_species)
        logit_d: Optional[torch.Tensor], # shape: (n_visits, n_species)
        center_logit_psi: Optional[torch.Tensor], # shape: (n_species,) or None if not on logits
        center_logit_d: Optional[torch.Tensor], # shape: (n_species,) or None if not on logits
        site_reduction: str,
        species_reduction: str,
    ) -> torch.Tensor:
    """
    Compute regularization penalty term.

    Parameters
    ----------
    penalization_type : str or None
        Penalty type among parameter- or logit-based options. If ``None``, no penalty is applied. Examples include:
        "l2_logits", "l1_logits", "l2_logits_naive", "l1_logits_naive", "l2_params", "l1_params".
    penalization_lambda_psi, penalization_lambda_d : float or None
        Penalty coefficients for occupancy and detection components.
    model : OccuModel
        Model containing parameters.
    logit_psi, logit_d : torch.Tensor or None
        Predicted logits.
    center_logit_psi, center_logit_d : torch.Tensor or None
        Reference logits for logit-based penalties.
    site_reduction, species_reduction : str
        Reduction modes.

    Returns
    -------
    torch.Tensor
        Scalar penalty value.

    Raises
    ------
    ValueError
        If a reduction mode or penalty type is invalid.
    """

    if penalization_type is None:
        return 0.0

    elif penalization_type in [
        "l2_logits", "l1_logits",
        "l2_logits_naive", "l1_logits_naive"
    ]:

        psi_diff = logit_psi - center_logit_psi.unsqueeze(0) # shape: (n_sites, n_species)
        d_diff   = logit_d   - center_logit_d.unsqueeze(0)   # shape: (n_visits, n_species)
        if penalization_type in ["l2_logits", "l2_logits_naive"]:
            penalty_psi, penalty_d = psi_diff ** 2, d_diff ** 2
        else: # "l1_logits" or "l1_logits_naive"
            penalty_psi, penalty_d = psi_diff.abs(), d_diff.abs()
        penalty_psi = penalty_psi.sum()
        penalty_d   = penalty_d.sum()

        if site_reduction == "mean":
            n_sites = logit_psi.shape[0]
            penalty_psi /= n_sites
            penalty_d   /= n_sites
        elif site_reduction == "sum":
            pass
        else:
            raise ValueError(f"Invalid site_reduction: {site_reduction}")
        if species_reduction == "mean":
            n_species = logit_psi.shape[1]
            penalty_psi /= n_species
            penalty_d   /= n_species
        elif species_reduction == "sum":
            pass
        else:
            raise ValueError(f"Invalid species_reduction: {species_reduction}")

        return (penalization_lambda_psi * penalty_psi) + \
               (penalization_lambda_d * penalty_d)

    elif penalization_type in ["l2_params", "l1_params"]:

        params_psi = torch.cat([p.view(-1) for p in model.psi_model.parameters()]) # shape: (n_params_psi,)
        params_d = torch.cat([p.view(-1) for p in model.d_model.parameters()]) # shape: (n_params_d,)

        if penalization_type == "l2_params":
            penalty_psi = params_psi.pow(2).sum()
            penalty_d   = params_d.pow(2).sum()
        else: # "l1_params"
            penalty_psi = params_psi.abs().sum()
            penalty_d   = params_d.abs().sum()

        return (penalization_lambda_psi * penalty_psi) + \
               (penalization_lambda_d * penalty_d)

    else:
        raise ValueError(f"Unsupported penalization type: {penalization_type}.")

    def nll_logits(
    y: torch.Tensor,
    *,
    logit_psi: torch.Tensor,
    logit_d: torch.Tensor,
    indices_visit_site: torch.Tensor,
    site_reduction: SiteReduction = "sum",
    species_reduction: SpeciesReduction = "mean",
    validate_inputs: bool = False,
) -> torch.Tensor:
    """
    Negative log-likelihood in logit space for single-season multi-species occupancy.

    Parameters
    ----------
    y : torch.Tensor
        Visit-level detections, shape ``(n_visits, n_species)``.
    logit_psi : torch.Tensor
        Occupancy logits, shape ``(n_sites, n_species)``.
    logit_d : torch.Tensor
        Detection logits, shape ``(n_visits, n_species)``.
    indices_visit_site : torch.Tensor
        Site index per visit.
    site_reduction : {"sum", "mean", "none"}, default="sum"
        Site-level reduction.
    species_reduction : {"mean", "sum", "none"}, default="mean"
        Species-level reduction.
    validate_inputs : bool, default=False
        Whether to validate input shapes, dtypes, and index consistency. 
        Can be set to False for performance if inputs are guaranteed to be correct 
        (e.g., when called from a training loop).

    Returns
    -------
    torch.Tensor
        NLL tensor with shape determined by reductions.
    """
    y = torch.as_tensor(y) # shape: (n_visits, n_species)
    logit_psi = torch.as_tensor(logit_psi) # shape: (n_sites, n_species)
    logit_d = torch.as_tensor(logit_d) # shape: (n_visits, n_species)
    indices_visit_site = torch.as_tensor(indices_visit_site) # shape: (n_visits,)

    if validate_inputs:
        n_sites, _, n_species = _validate_inputs(y, logit_psi, logit_d, indices_visit_site)
    else:
        n_sites, n_species = logit_psi.shape

    device = y.device
    y_dtype = y.dtype

    # Binary indicator of whether each species was detected at each site
    detected = torch.zeros((n_sites, n_species), device=device, dtype=y_dtype)
    detected.index_add_(0, indices_visit_site, y)
    detected = (detected > 0).to(y_dtype) # shape: (n_sites, n_species)

    # NB: the following tricks are used for numerical stability (if using sigmoid loss function and logits):
    #  log(prob) = log(1 / (1 + exp(-logit))) = LogSigmoid(logit)
    #  log(1 - prob) = log(1 - 1 / (1 + exp(-logit))) = log( 1 / (1 + exp(logit)) ) = LogSigmoid(-logit)

    # Term for sites with detections: sum_s [ - log(psi_s) - sum_v ( y_v log(d_v) + (1 - y_v) log(1 - d_v) ) ] = sum_s( - log(psi_s) + sum_v BCE(d_v, y_v) )
    nll_det = -F.logsigmoid(logit_psi) # = -log(psi); shape: (n_sites, n_species)
    bce = F.binary_cross_entropy_with_logits(logit_d, y, reduction="none") # shape: (n_visits, n_species)
    nll_det.index_add_(0, indices_visit_site, bce) # Accumulate BCE terms of the species for each site; shape: (n_sites, n_species)

    # Term for sites with no detections: sum_s [ - log(1 - psi_s + psi_s * prod_v (1 - d_v) ) ]
    log_prod_1m_d = torch.zeros((n_sites, n_species), device=device, dtype=logit_d.dtype) # shape: (n_sites, n_species)
    log_prod_1m_d.index_add_(0, indices_visit_site, F.logsigmoid(-logit_d)) # Accumulate log(1 - d_v) terms of the species for each site; shape: (n_sites, n_species)

    log1mpsi = F.logsigmoid(-logit_psi) # = log(1 - psi); shape: (n_sites, n_species)
    logpsi = F.logsigmoid(logit_psi) # = log(psi); shape: (n_sites, n_species)
    nll_no_det = -torch.logaddexp(log1mpsi, logpsi + log_prod_1m_d) # = -log(1 - psi + psi * prod_v (1 - d_v)); shape: (n_sites, n_species)

    # Combine terms based on whether species was detected or not at each site
    nll_terms = detected * nll_det + (1.0 - detected) * nll_no_det  # (n_sites, n_species)

    # Reduce over species (dim=1)
    if species_reduction == "none":
        out = nll_terms # shape: (n_sites, n_species)
    elif species_reduction == "mean":
        out = nll_terms.mean(dim=1)  # shape: (n_sites,)
    elif species_reduction == "sum":
        out = nll_terms.sum(dim=1)   # shape: (n_sites,)
    else:
        raise ValueError("species_reduction must be 'none', 'mean', or 'sum'.")

    # Reduce over sites
    if site_reduction == "none":
        return out
    if site_reduction == "sum":
        return out.sum(dim=0)
    if site_reduction == "mean":
        return out.mean(dim=0)
    raise ValueError("site_reduction must be 'none', 'mean', or 'sum'.")

    torch.Tensor

    logit_d = torch.log(d) - torch.log1p(-d)

        y,
        logit_psi=logit_psi,
        logit_d=logit_d,
        indices_visit_site=indices_visit_site,
        site_reduction=site_reduction,
        species_reduction=species_reduction,
        validate_inputs=validate_inputs

    def initialize_rough_estimate(self, data: "OccuData") -> None:
        """
        Initialize intercept logits from rough moment-based estimates.

        Parameters
        ----------
        data : OccuData
            Occupancy dataset used to compute rough estimates.

        Returns
        -------
        None
        """
        y = data.y
        indices_visit_site = data.indices_visit_site # shape: (n_visits,) with values in [0, n_sites-1]

        n_sites = data.n_sites
        n_visits = data.n_visits
        n_species = data.n_species

        n_D = y.sum(dim=0) # shape: (n_species,)
        S = n_sites # total number of sites
        K = n_visits / n_sites # average number of visits per site
        # Number sites with at least one detection for each species

        # Binary indicator of whether each species was detected at each site
        detected = torch.zeros((n_sites, n_species), device=y.device, dtype=y.dtype)
        detected.index_add_(0, indices_visit_site, y)
        detected = detected.bool() # shape: (n_sites, n_species)
        S_D = detected.sum(dim=0) # shape: (n_species,)

        # Solve the equations for the rough estimates of psi and d. 
        # The equations are:
        # hat_d = 1 - (1 - hat_d) ** K * n_D / (S_D * K)
        # hat_psi = S_D / (S * (1 - (1 - hat_d) ** K))

        # If we approximate 1 - (1 - hat_d) ** K by 1
        hat_d = (n_D / (S_D * K)).clamp(min=1e-6, max=1-1e-6)
        hat_psi = (S_D / (S * (1 - (1 - hat_d) ** K))).clamp(min=1e-6, max=1-1e-6)

        with torch.no_grad():
            self.psi_model.logit_species.copy_(torch.logit(hat_psi))
            self.d_model.logit_species.copy_(torch.logit(hat_d))
